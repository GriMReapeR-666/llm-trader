The Critical Operational Flaw Flagged
The official MetaTrader5 Python library (import MetaTrader5 as mt5) communicates with terminal64.exe over a single-threaded Windows Named Pipe.
In operations/dual_velocity.py, your Fast Lane polls mt5.symbol_info_tick() and mt5.positions_get() every sub-second, while your Slow Lane (ContinuousBacktest, AlphaDiscovery, daily_reconciliation.py) calls mt5.copy_rates_from_pos() and mt5.history_deals_get().
If both lanes call mt5.* functions simultaneously from different Python threads without a global re-entrant lock, the Windows Named Pipe buffer interleaves responses, returning None or throwing IPC timeout (-10005), which can cause the Fast Lane to miss a trailing-stop update or think a position closed when it didn't.
Furthermore, if terminal64.exe briefly loses connection to your broker server (terminal_info().connected == False), unguarded orders fail silently.

import threading
import time

class MT5SafeGateway:
    """
    Serializes all Fast-Lane and Slow-Lane calls over MetaTrader5's single-threaded
    Windows Named Pipe and automatically recovers from IPC drops (-10004 / -10005).
    """
    _lock = threading.RLock()

    def __init__(self, mt5_module, login: int | None = None, password: str | None = None, server: str | None = None, path: str | None = None):
        self.mt5 = mt5_module
        self.login = login
        self.password = password
        self.server = server
        self.path = path

    def ensure_connected(self) -> bool:
        with self._lock:
            term = self.mt5.terminal_info()
            if term is not None and getattr(term, "connected", False):
                return True

            # Attempt clean re-initialization of the IPC pipe
            self.mt5.shutdown()
            time.sleep(0.5)
            init_kwargs = {}
            if self.path:
                init_kwargs["path"] = self.path
            if self.login and self.password and self.server:
                init_kwargs.update({"login": self.login, "password": self.password, "server": self.server})

            if not self.mt5.initialize(**init_kwargs):
                return False

            term = self.mt5.terminal_info()
            return bool(term is not None and getattr(term, "connected", False))

    def call(self, fn_name: str, *args, **kwargs):
        """
        Thread-safe serialized IPC call with automatic retry on transient pipe drop.
        """
        with self._lock:
            if not self.ensure_connected():
                return None
            fn = getattr(self.mt5, fn_name)
            result = fn(*args, **kwargs)
            if result is None:
                err_code, _ = self.mt5.last_error()
                # -10004 / -10005 = IPC connection lost; reconnect once and retry
                if err_code in (-10004, -10005, -10001):
                    if self.ensure_connected():
                        result = fn(*args, **kwargs)
            return result

def evaluate_macro_event_shield(
    symbol: str,
    upcoming_events: list[dict],  # [{"currency": "USD", "impact": "HIGH", "timestamp_utc": float, "title": str}]
    utc_now_ts: float | None = None,
    pre_event_freeze_mins: int = 15,
    post_event_freeze_mins: int = 10,
) -> dict:
    """
    Protects $100 accounts around High-Impact (Red-Folder) macro releases:
    1. Blocks new entries from -15m before to +10m after any HIGH impact event on base or quote currency.
    2. Signals trade_manager.py to ratchet open profitable positions to Break-Even 5 minutes before release.
    """
    now = utc_now_ts or datetime.now(timezone.utc).timestamp()
    base_ccy = symbol[:3].upper()
    quote_ccy = symbol[3:6].upper()
    relevant_ccys = {base_ccy, quote_ccy, "USD"}  # USD events impact all FX pairs

    for ev in upcoming_events:
        if ev.get("impact", "").upper() != "HIGH":
            continue
        if ev.get("currency", "").upper() not in relevant_ccys:
            continue

        ev_ts = float(ev["timestamp_utc"])
        mins_to_event = (ev_ts - now) / 60.0

        # Inside [-15m, +10m] blackout window
        if -post_event_freeze_mins <= mins_to_event <= pre_event_freeze_mins:
            tighten_open = 0.0 <= mins_to_event <= 5.0
            return {
                "allow_new_entries": False,
                "tighten_open_stops_to_be": tighten_open,
                "active_event": ev.get("title", "HIGH_IMPACT_MACRO"),
                "mins_to_event": round(mins_to_event, 1),
            }

    return {
        "allow_new_entries": True,
        "tighten_open_stops_to_be": False,
        "active_event": None,
        "mins_to_event": None,
    }

import numpy as np

def run_strategy_autopsy(trades: list[dict]) -> dict:
    """
    Forensic Quantitative Autopsy Engine:
    Takes a list of trade dicts:
      {
        "r_multiple": float,
        "mfe_r": float,            # Max Favorable Excursion in R units during trade
        "mae_r": float,            # Max Adverse Excursion in R units during trade
        "post_sl_reached_1_5r": bool, # Did price hit +1.5R within 6 bars after stopping out?
        "regime": str,             # e.g., "LOW_VOL_MEAN_REVERT", "TREND_EXPANSION"
        "hour_utc": int,           # 0..23
        "gross_pnl": float,
        "friction_cost": float,    # spread + commission + swap in dollars
      }
    Returns a structured Root-Cause Diagnosis JSON for the LLM Reasoner & StrategyGit.
    """
    if len(trades) < 20:
        return {"status": "INSUFFICIENT_DATA", "diagnoses": [], "recommended_mutations": []}

    r_arr = np.array([float(t["r_multiple"]) for t in trades])
    mfe_arr = np.array([float(t.get("mfe_r", max(0.0, t["r_multiple"]))) for t in trades])
    mae_arr = np.array([float(t.get("mae_r", abs(min(0.0, t["r_multiple"])))) for t in trades])

    losers = [t for t in trades if float(t["r_multiple"]) < 0]
    winners = [t for t in trades if float(t["r_multiple"]) > 0]

    diagnoses = []
    mutations = []

    # 1. Check Inverted Signal Polarity
    if np.mean(r_arr) < -0.15 and np.mean(mae_arr) > 1.4 * max(np.mean(mfe_arr), 1e-4):
        diagnoses.append({
            "code": "INVERTED_SIGNAL_POLARITY",
            "severity": "CRITICAL",
            "evidence": f"Mean R={np.mean(r_arr):.2f}R, Mean MAE={np.mean(mae_arr):.2f}R vs MFE={np.mean(mfe_arr):.2f}R",
        })
        mutations.append({"op": "INVERT_ENTRY_SIGNAL", "reason": "Strong adverse excursion dominates favorable excursion"})

    # 2. Check Premature Stop-Out (Stop Hunt)
    if losers:
        stop_hunted_rate = np.mean([1.0 if t.get("post_sl_reached_1_5r", False) else 0.0 for t in losers])
        if stop_hunted_rate >= 0.40:
            diagnoses.append({
                "code": "PREMATURE_STOP_OUT",
                "severity": "HIGH",
                "evidence": f"{stop_hunted_rate*100:.1f}% of stopped-out trades subsequently rallied to +1.5R",
            })
            mutations.append({"op": "WIDEN_SL_ATR_MULT", "scale": 1.30, "reason": "Stops placed inside noise envelope"})

    # 3. Check Profit Give-Back (Exits too greedy / slow ratchet)
    if losers:
        giveback_rate = np.mean([1.0 if float(t.get("mfe_r", 0.0)) >= 1.2 else 0.0 for t in losers])
        if giveback_rate >= 0.40:
            diagnoses.append({
                "code": "PROFIT_GIVEBACK_BLEED",
                "severity": "HIGH",
                "evidence": f"{giveback_rate*100:.1f}% of losers first reached >= +1.2R MFE before reversing to a loss",
            })
            mutations.append({"op": "LOWER_BREAKEVEN_TRIGGER_R", "new_trigger_r": 0.95, "reason": "Lock in +1R excursions earlier"})

    # 4. Check Regime Contamination Matrix
    regime_stats = {}
    for t in trades:
        reg = t.get("regime", "UNKNOWN")
        regime_stats.setdefault(reg, []).append(float(t["r_multiple"]))

    good_regimes = [reg for reg, rs in regime_stats.items() if len(rs) >= 8 and np.mean(rs) >= 0.25]
    bad_regimes = [reg for reg, rs in regime_stats.items() if len(rs) >= 8 and np.mean(rs) <= -0.15]

    if good_regimes and bad_regimes:
        diagnoses.append({
            "code": "REGIME_CONTAMINATION",
            "severity": "CRITICAL",
            "evidence": {reg: {"n": len(rs), "mean_r": round(float(np.mean(rs)), 3)} for reg, rs in regime_stats.items()},
        })
        mutations.append({
            "op": "RESTRICT_ALLOWED_REGIMES",
            "allowed_regimes": good_regimes,
            "blocked_regimes": bad_regimes,
            "reason": f"Profitable in {good_regimes}, bleeding in {bad_regimes}",
        })

    gross_wins = sum(max(0.0, float(t.get("gross_pnl", 0.0))) for t in winners)
    total_friction = sum(float(t.get("friction_cost", 0.0)) for t in trades)
    if gross_wins > 0 and (total_friction / gross_wins) >= 0.35:

import numpy as np

def verify_strategy_math_and_causality(
    strategy_signal_fn,
    ohlcv_bars: np.ndarray,
    trades_ledger: list[dict],
    initial_equity: float = 100.0,
    terminal_equity: float = 100.0,
) -> dict:
    """
    Deterministic Mathematical & Causality Auditor:
    1. THE TIME-MACHINE TEST: Proves 0.000000000000 lookahead bias by corrupting future bars
       and verifying historical signals at bar t remain bit-for-bit identical.
    2. FINITE DOMAIN CHECK: Guarantees signals contain zero NaN/Inf values.
    3. ACCOUNTING CONSERVATION IDENTITY: Verifies sum(net_pnl) == (terminal_equity - initial_equity)
       and r_multiple == net_pnl / risk_dollars within 1e-5.
    """
    n = len(ohlcv_bars)
    if n < 100:
        return {"passed": False, "reason": "Insufficient bars for causality audit"}

    # 1. Time-Machine Causality Perturbation Test at 3 checkpoints
    full_signals = np.asarray(strategy_signal_fn(ohlcv_bars), dtype=float)
    if not np.all(np.isfinite(full_signals)):
        return {"passed": False, "reason": "MATH_ERROR: Strategy emitted NaN or Inf signals"}

    for t_check in (n // 2, int(n * 0.75), n - 15):
        # Feed strictly truncated history up to t_check
        past_only_bars = ohlcv_bars[: t_check + 1].copy()
        sig_past_only = float(strategy_signal_fn(past_only_bars)[-1])

        # Now feed full bars where future bars (t_check+1 .. n) are corrupted by 100x spikes
        corrupted_future_bars = ohlcv_bars.copy()
        corrupted_future_bars[t_check + 1 :] *= 100.0
        sig_with_corrupted_future = float(strategy_signal_fn(corrupted_future_bars)[t_check])

        if abs(sig_past_only - sig_with_corrupted_future) > 1e-12 or abs(sig_past_only - float(full_signals[t_check])) > 1e-12:
            return {
                "passed": False,
                "reason": (
                    f"LOOKAHEAD_BIAS_DETECTED at bar {t_check}: "
                    f"past_only={sig_past_only:.8f} vs future_corrupted={sig_with_corrupted_future:.8f}"
                ),
            }

    # 2. Accounting & R-Multiple Conservation Check
    total_net_pnl = 0.0
    for idx, tr in enumerate(trades_ledger):
        pnl = float(tr["net_pnl"])
        risk = float(tr["risk_dollars"])
        r_mult = float(tr["r_multiple"])
        if risk <= 0:
            return {"passed": False, "reason": f"MATH_ERROR: Trade {idx} has non-positive risk_dollars={risk}"}
        expected_r = pnl / risk
        if abs(r_mult - expected_r) > 1e-4:
            return {
                "passed": False,
                "reason": f"MATH_ERROR: Trade {idx} R-multiple mismatch ({r_mult:.4f} != {pnl:.2f}/{risk:.2f}={expected_r:.4f})",
            }
        total_net_pnl += pnl

    equity_delta = terminal_equity - initial_equity
    if abs(total_net_pnl - equity_delta) > 1e-2:
        return {
            "passed": False,
            "reason": f"ACCOUNTING_MISMATCH: sum(net_pnl)={total_net_pnl:.2f} != delta_equity={equity_delta:.2f}",
        }

    return {"passed": True, "reason": "ALL_CAUSALITY_AND_MATH_INVARIANTS_VERIFIED"}

import hashlib
import json
import sqlite3
import time
import numpy as np
from scipy.stats import wilcoxon

class StrategyGit:
    """
    Git-like Version Control System for Autonomous Quantitative Strategies.
    Supports: commit(), branch(), diff(), merge_pull_request(), and revert().
    """
    def __init__(self, db_conn: sqlite3.Connection):
        self.db = db_conn
        self._init_tables()

    def _init_tables(self) -> None:
        cur = self.db.cursor()
        cur.execute(
            """
            CREATE TABLE IF NOT EXISTS strategy_commits (
                commit_hash TEXT PRIMARY KEY,
                strategy_id TEXT NOT NULL,
                parent_hash TEXT,
                branch_name TEXT NOT NULL,
                genome_json TEXT NOT NULL,
                autopsy_Diagnosis_json TEXT,
                commit_message TEXT NOT NULL,
                cpcv_oos_sharpes_json TEXT NOT NULL,
                mean_sharpe REAL NOT NULL,
                pbo REAL NOT NULL,
                dsr REAL NOT NULL,
                math_audit_passed INTEGER NOT NULL,
                created_at REAL NOT NULL
            )
            """
        )
        cur.execute(
            """
            CREATE TABLE IF NOT EXISTS strategy_branches (
                strategy_id TEXT NOT NULL,
                branch_name TEXT NOT NULL,
                head_commit_hash TEXT NOT NULL,
                updated_at REAL NOT NULL,
                PRIMARY KEY (strategy_id, branch_name)
            )
            """
        )
        self.db.commit()

    @staticmethod
    def compute_commit_hash(strategy_id: str, genome: dict, parent_hash: str | None) -> str:
        payload = json.dumps(
            {"strategy_id": strategy_id, "genome": genome, "parent": parent_hash},
            sort_keys=True,
        ).encode("utf-8")
        return hashlib.sha256(payload).hexdigest()[:12]

    def commit(
        self,
        strategy_id: str,
        branch_name: str,
        genome: dict,
        commit_message: str,
        cpcv_oos_sharpes: list[float],
        pbo: float,
        dsr: float,
        math_audit_passed: bool,
        parent_hash: str | None = None,
        autopsy_diagnosis: dict | None = None,
    ) -> dict:
        if not math_audit_passed:
            return {"committed": False, "reason": "REJECTED: Failed core/math_auditor.py verification"}

        commit_hash = self.compute_commit_hash(strategy_id, genome, parent_hash)
        mean_sr = float(np.mean(cpcv_oos_sharpes)) if cpcv_oos_sharpes else 0.0
        now = time.time()

        cur = self.db.cursor()
        cur.execute(
            """
            INSERT OR REPLACE INTO strategy_commits (
                commit_hash, strategy_id, parent_hash, branch_name, genome_json,
                autopsy_diagnosis_json, commit_message, cpcv_oos_sharpes_json,
                mean_sharpe, pbo, dsr, math_audit_passed, created_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """,
            (
                commit_hash,
                strategy_id,
                parent_hash,
                branch_name,
                json.dumps(genome, sort_keys=True),
                json.dumps(autopsy_diagnosis or {}),
                commit_message,
                json.dumps([round(float(x), 5) for x in cpcv_oos_sharpes]),
                mean_sr,
                float(pbo),
                float(dsr),
                1 if math_audit_passed else 0,
                now,
            ),
        )
        cur.execute(
            """
            INSERT OR REPLACE INTO strategy_branches (strategy_id, branch_name, head_commit_hash, updated_at)
            VALUES (?, ?, ?, ?)
            """,
            (strategy_id, branch_name, commit_hash, now),
        )
        self.db.commit()
        return {"committed": True, "commit_hash": commit_hash, "branch": branch_name, "mean_sharpe": mean_sr}

    def diff(self, commit_a: str, commit_b: str) -> dict:
        """
        Computes structural genome diff and quantitative metric delta between two commits.
        """
        cur = self.db.cursor()
        row_a = cur.execute("SELECT * FROM strategy_commits WHERE commit_hash = ?", (commit_a,)).fetchone()
        row_b = cur.execute("SELECT * FROM strategy_commits WHERE commit_hash = ?", (commit_b,)).fetchone()
        if not row_a or not row_b:
            return {"error": "Commit not found"}

        gen_a = json.loads(row_a["genome_json"])
        gen_b = json.loads(row_b["genome_json"])
        all_keys = sorted(set(gen_a.keys()) | set(gen_b.keys()))
        param_diffs = {
            k: {"from": gen_a.get(k), "to": gen_b.get(k)}
            for k in all_keys
            if gen_a.get(k) != gen_b.get(k)
        }
        return {
            "from_commit": commit_a,
            "to_commit": commit_b,
            "genome_changes": param_diffs,
            "sharpe_delta": round(float(row_b["mean_sharpe"]) - float(row_a["mean_sharpe"]), 4),
            "dsr_delta": round(float(row_b["dsr"]) - float(row_a["dsr"]), 4),
            "pbo_delta": round(float(row_b["pbo"]) - float(row_a["pbo"]), 4),
        }

    def merge_pull_request(
        self,
        strategy_id: str,
        source_branch: str,
        target_branch: str = "canary",
        max_p_val: float = 0.05,
        min_dsr: float = 0.95,
        max_pbo: float = 0.25,
    ) -> dict:
        """
        Statistical Pull-Request Gate:
        Merges an experimental mutation branch (e.g. 'shadow/fix-regime') into 'canary' or 'main'
        ONLY IF it beats the current target branch HEAD on Paired Wilcoxon Signed-Rank Test (p < 0.05)
        across the 15 CPCV out-of-sample splits AND passes DSR/PBO gates.
        """
        cur = self.db.cursor()
        src_head = cur.execute(
            "SELECT head_commit_hash FROM strategy_branches WHERE strategy_id=? AND branch_name=?",
            (strategy_id, source_branch),
        ).fetchone()
        tgt_head = cur.execute(
            "SELECT head_commit_hash FROM strategy_branches WHERE strategy_id=? AND branch_name=?",
            (strategy_id, target_branch),
        ).fetchone()

        if not src_head:
            return {"merged": False, "reason": f"Source branch {source_branch} not found"}

        cand = cur.execute("SELECT * FROM strategy_commits WHERE commit_hash=?", (src_head[0],)).fetchone()
        if not cand["math_audit_passed"] or float(cand["dsr"]) < min_dsr or float(cand["pbo"]) > max_pbo:
            return {
                "merged": False,
                "reason": f"Candidate {src_head[0]} failed hard gates (DSR={cand['dsr']:.3f}, PBO={cand['pbo']:.3f})",
            }

        # If target branch already exists, require Paired Wilcoxon Superiority over current target HEAD
        if tgt_head:
            base = cur.execute("SELECT * FROM strategy_commits WHERE commit_hash=?", (tgt_head[0],)).fetchone()
            cand_srs = np.array(json.loads(cand["cpcv_oos_sharpes_json"]), dtype=float)
            base_srs = np.array(json.loads(base["cpcv_oos_sharpes_json"]), dtype=float)

            if len(cand_srs) == len(base_srs) and len(cand_srs) >= 6:
                _, p_val = wilcoxon(cand_srs, base_srs, alternative="greater")
                if p_val > max_p_val:
                    return {
                        "merged": False,
                        "reason": f"Mutation not statistically superior to {tgt_head[0]} (Wilcoxon p={p_val:.4f} > {max_p_val})",
                    }

        # Fast-forward target branch pointer to candidate commit
        cur.execute(
            """
            INSERT OR REPLACE INTO strategy_branches (strategy_id, branch_name, head_commit_hash, updated_at)
            VALUES (?, ?, ?, ?)
            """,
            (strategy_id, target_branch, src_head[0], time.time()),
        )
        self.db.commit()
        return {"merged": True, "strategy_id": strategy_id, "target_branch": target_branch, "new_head": src_head[0]}

    def revert_to_parent(self, strategy_id: str, branch_name: str = "main") -> dict:
        """
        Autonomous `git revert`: Called by DegradationDetector when a live commit triggers CUSUM drift.
        Atomically rolls the branch HEAD back to its parent_hash.
        """
        cur = self.db.cursor()
        head = cur.execute(
            "SELECT head_commit_hash FROM strategy_branches WHERE strategy_id=? AND branch_name=?",
            (strategy_id, branch_name),
        ).fetchone()
        if not head:
            return {"reverted": False, "reason": "Branch not found"}

        curr_commit = cur.execute("SELECT parent_hash FROM strategy_commits WHERE commit_hash=?", (head[0],)).fetchone()
        if not curr_commit or not curr_commit["parent_hash"]:
            return {"reverted": False, "reason": "No parent commit to revert to; quarantine strategy instead"}

        parent_hash = curr_commit["parent_hash"]
        cur.execute(
            "UPDATE strategy_branches SET head_commit_hash=?, updated_at=? WHERE strategy_id=? AND branch_name=?",
            (parent_hash, time.time(), strategy_id, branch_name),
        )
        self.db.commit()
        return {"reverted": True, "strategy_id": strategy_id, "from_commit": head[0], "reverted_to": parent_hash}

import copy
import json

class StrategyReasoner:
    """
    Hybrid Causal Strategy Reasoner & Alpha-Grafting Engine.
    1. Synthesizes autopsy telemetry + StrategyGit commit history (avoiding previously failed patches).
    2. Queries LLMRouter for causal hypothesis generation when available.
    3. Falls back to Deterministic Autopsy-to-Genome compilation + FactorMiner Alpha-Grafting
       when LLMs are offline or when a strategy's core entry signal is dead across all regimes.
    """
    def __init__(self, llm_router=None):
        self.llm_router = llm_router

    def propose_improvement_branches(
        self,
        strategy_id: str,
        parent_genome: dict,
        autopsy_report: dict,
        failed_commit_Diffs: list[dict],
        validated_alphas_pool: list[dict] | None = None,
    ) -> list[dict]:
        """
        Returns 1 to 3 candidate mutation branches:
          [{"branch_name": str, "mutated_genome": dict, "commit_message": str}]
        """
        candidates = []
        base = copy.deepcopy(parent_genome)
        recommended = autopsy_report.get("recommended_mutations", [])

        # 1. Deterministic Structural Repair from Forensic Autopsy
        if recommended:
            mutated = copy.deepcopy(base)
            applied_ops = []

            for rec in recommended:
                op = rec.get("op")
                if op == "INVERT_ENTRY_SIGNAL":
                    mutated["signal_polarity"] = -1 * int(mutated.get("signal_polarity", 1))
                    applied_ops.append("invert-polarity")

                elif op == "WIDEN_SL_ATR_MULT":
                    old_sl = float(mutated.get("sl_atr_mult", 1.5))
                    mutated["sl_atr_mult"] = round(min(old_sl * float(rec.get("scale", 1.3)), 3.5), 2)
                    applied_ops.append("widen-sl")

                elif op == "LOWER_BREAKEVEN_TRIGGER_R":
                    mutated["breakeven_trigger_r"] = float(rec.get("new_trigger_r", 0.95))
                    applied_ops.append("early-be-ratchet")

                elif op == "RESTRICT_ALLOWED_REGIMES":
                    mutated["allowed_regimes"] = list(rec.get("allowed_regimes", []))
                    applied_ops.append("regime-lock")

                elif op == "INCREASE_MIN_TARGET_ATR":
                    old_tp = float(mutated.get("tp_atr_mult", 2.5))
                    mutated["tp_atr_mult"] = round(min(old_tp * float(rec.get("scale", 1.4)), 6.0), 2)
                    applied_ops.append("widen-tp-friction")

            if applied_ops and mutated != base:
                branch_slug = "-".join(applied_ops[:2])
                candidates.append({
                    "branch_name": f"shadow/fix-{branch_slug}",
                    "mutated_genome": mutated,
                    "commit_message": f"Autopsy repair [{', '.join(applied_ops)}] on {strategy_id}",
                })

        # 2. Structural Alpha-Grafting (when core signal is weak across ALL regimes)
        # If overall mean R < 0 and no single regime had strong positive expectancy, graft a top FactorMiner alpha
        has_regime_rescue = any(r.get("op") == "RESTRICT_ALLOWED_REGIMES" for r in recommended)
        if (not has_regime_rescue or autopsy_report.get("overall_mean_r", 0.0) < 0.05) and validated_alphas_pool:
            # Sort validated alphas by highest IC_IR and low turnover
            ranked_alphas = sorted(
                validated_alphas_pool,
                key=lambda a: float(a.get("ic_ir", 0.0)) * (1.0 - 0.3 * float(a.get("turnover", 0.5))),
                reverse=True,
            )
            for idx, best_alpha in enumerate(ranked_alphas[:2]):
                graft_genome = copy.deepcopy(base)
                graft_genome["primary_alpha_dsl"] = best_alpha["expression"]
                graft_genome["signal_polarity"] = 1  # Polarity already normalized by Pillar 2
                candidates.append({
                    "branch_name": f"shadow/graft-alpha-{idx+1}",
                    "mutated_genome": graft_genome,
                    "commit_message": (
                        f"Grafted FactorMiner alpha (IC_IR={best_alpha.get('ic_ir', 0):.2f}): "
                        f"{best_alpha['expression'][:45]}"
                    ),
                })

        # 3. Optional LLM Causal Refinement Branch (if LLMRouter is online)
        if self.llm_router is not None:
            llm_branch = self._query_llm_mutation(strategy_id, base, autopsy_report, failed_commit_diffs)
            if llm_branch is not None:
                candidates.append(llm_branch)

        return candidates[:3]

    def _query_llm_mutation(
        self,
        strategy_id: str,
        parent_genome: dict,
        autopsy_report: dict,
        failed_commit_diffs: list[dict],
    ) -> dict | None:
        prompt = (
            f"Strategy '{strategy_id}' failed validation.\n"
            f"Current Genome: {json.dumps(parent_genome)}\n"
            f"Forensic Autopsy: {json.dumps(autopsy_report)}\n"
            f"Previously Rejected Diffs (DO NOT REPEAT): {json.dumps(failed_commit_diffs[-5:])}\n"
            "Return ONLY a JSON object with keys: 'mutated_genome' (matching genome schema) and 'commit_message'."
        )
        try:
            resp = self.llm_router.generate_json(agent="strategy_reasoner", prompt=prompt)
            if isinstance(resp, dict) and "mutated_genome" in resp and resp["mutated_genome"] != parent_genome:
                return {
                    "branch_name": "shadow/llm-causal-patch",
                    "mutated_genome": resp["mutated_genome"],
                    "commit_message": str(resp.get("commit_message", "LLM causal genome patch"))[:120],
                }
        except Exception:

import numpy as np

class ShadowTournamentArena:
    """
    Runs multiple candidate StrategyGit branches side-by-side in real-time shadow mode
    and selects the statistically dominant branch to merge into 'canary'.
    """
    def __init__(self, min_shadow_trades: int = 15):
        self.min_shadow_trades = min_shadow_trades
        # {(strategy_id, branch_name): [r_multiple_1, r_multiple_2, ...]}
        self.shadow_ledgers: dict[tuple[str, str], list[float]] = {}

    def record_shadow_trade(self, strategy_id: str, branch_name: str, net_r_multiple: float) -> None:
        key = (strategy_id, branch_name)
        self.shadow_ledgers.setdefault(key, []).append(float(net_r_multiple))

    def evaluate_tournament(self, strategy_id: str, candidate_branches: list[str]) -> dict:
        """
        Compares competing shadow branches for strategy_id using Monte Carlo Thompson
        Probability of Best Arm and prunes branches with negative expectancy.
        """
        eligible = {}
        pruned = []

        for br in candidate_branches:
            r_list = self.shadow_ledgers.get((strategy_id, br), [])
            if len(r_list) < self.min_shadow_trades:
                continue
            r_arr = np.array(r_list, dtype=float)
            mean_r = float(np.mean(r_arr))
            if mean_r <= 0.0:
                pruned.append(br)
            else:
                eligible[br] = r_arr

        if not eligible:
            return {"winner_branch": None, "pruned_branches": pruned, "status": "AWAITING_SHADOW_SAMPLES"}

        # Bootstrap probability that each branch has the highest true Sharpe Ratio
        n_boot = 2000
        boot_scores = {}
        for br, r_arr in eligible.items():
            idx = np.random.randint(0, len(r_arr), size=(n_boot, len(r_arr)))
            samples = r_arr[idx]
            boot_sr = np.mean(samples, axis=1) / np.maximum(np.std(samples, axis=1, ddof=1), 1e-6)
            boot_scores[br] = boot_sr

        branch_names = list(boot_scores.keys())
        score_matrix = np.column_stack([boot_scores[b] for b in branch_names])
        winner_indices = np.argmax(score_matrix, axis=1)

        win_probs = {
            branch_names[i]: float(np.mean(winner_indices == i))
            for i in range(len(branch_names))
        }
        best_branch = max(win_probs, key=win_probs.get)

        if win_probs[best_branch] >= 0.70:
            return {
                "winner_branch": best_branch,
                "win_probability": round(win_probs[best_branch], 3),
                "pruned_branches": pruned,
                "status": "TOURNAMENT_WINNER_DECIDED",
            }

        return {
            "winner_branch": None,
            "win_probabilities": win_probs,
            "pruned_branches": pruned,
            "status": "TOURNAMENT_IN_PROGRESS",
        }

import time
import numpy as np

def run_autonomous_strategy_evolution_sprint(
    strategy_id: str,
    current_genome: dict,
    parent_commit_hash: str,
    parent_trades: list[dict],
    ohlcv_bars: np.ndarray,
    backtest_runner_fn,          # fn(genome, ohlcv_bars) -> (trades_list, signal_fn)
    strategy_git,                # StrategyGit instance
    strategy_reasoner,           # StrategyReasoner instance
    validated_alphas_pool: list[dict],
    n_effective_trials: float = 13.0,
) -> dict:
    """
    End-to-End Self-Healing Strategy Evolution Pipeline:
    Autopsy -> Reason/Graft -> Math & Causality Audit -> CPCV & DSR -> StrategyGit Commit & PR Merge.
    """
    from research.strategy_autopsy import run_strategy_autopsy
    from core.math_auditor import verify_strategy_math_and_causality

    # Step 1: Forensic Autopsy on the weak parent strategy
    autopsy = run_strategy_autopsy(parent_trades)

    # Step 2: Generate targeted improvement branches (structural fix + alpha grafting + LLM patch)
    candidate_branches = strategy_reasoner.propose_improvement_branches(
        strategy_id=strategy_id,
        parent_genome=current_genome,
        autopsy_report=autopsy,
        failed_commit_diffs=[],
        validated_alphas_pool=validated_alphas_pool,
    )

    sprint_results = []
    for cand in candidate_branches:
        # Yield GIL briefly so Fast-Lane MT5 tick monitoring never lags
        time.sleep(0.002)

        branch_name = cand["branch_name"]
        mutated_genome = cand["mutated_genome"]
        commit_msg = cand["commit_message"]

        # Step 3: Execute backtest on mutated genome
        child_trades, child_signal_fn = backtest_runner_fn(mutated_genome, ohlcv_bars)
        if len(child_trades) < 60:
            sprint_results.append({"branch": branch_name, "status": "REJECTED_TOO_FEW_TRADES"})
            continue

        # Step 4: Self-Verifying Math & Zero-Lookahead Causality Audit
        net_pnls = sum(float(t["net_pnl"]) for t in child_trades)
        math_audit = verify_strategy_math_and_causality(
            strategy_signal_fn=child_signal_fn,
            ohlcv_bars=ohlcv_bars,
            trades_ledger=child_trades,
            initial_equity=100.0,
            terminal_equity=100.0 + net_pnls,
        )
        if not math_audit["passed"]:
            sprint_results.append({"branch": branch_name, "status": "REJECTED_MATH_AUDIT", "reason": math_audit["reason"]})
            continue

        # Step 5: Stage 3b Purged CPCV & Stage 5 Eigenvalue DSR Evaluation
        r_multiples = np.array([float(t["r_multiple"]) for t in child_trades], dtype=float)
        cpcv_res = run_stage_3b_cpcv(r_multiples)
        # Account for the new mutation trial in N_eff
        n_eff_updated = n_effective_trials + 0.5
        dsr_res = compute_deflated_sharpe_ratio(r_multiples, n_eff=n_eff_updated)

        # Step 6: Commit to StrategyGit
        commit_res = strategy_git.commit(
            strategy_id=strategy_id,
            branch_name=branch_name,
            genome=mutated_genome,
            commit_message=commit_msg,
            cpcv_oos_sharpes=cpcv_res.get("oos_sharpes", [cpcv_res["mean_sharpe"]] * 15),
            pbo=cpcv_res["pbo"],
            dsr=dsr_res["dsr"],
            math_audit_passed=True,
            parent_hash=parent_commit_hash,
            autopsy_diagnosis=autopsy,
        )

        # Step 7: Attempt Statistical Pull-Request Merge into 'canary'
        merge_res = strategy_git.merge_pull_request(
            strategy_id=strategy_id,
            source_branch=branch_name,
            target_branch="canary",
            max_p_val=0.05,
            min_dsr=0.95,
            max_pbo=0.25,
        )
        sprint_results.append({
            "branch": branch_name,
            "commit_hash": commit_res.get("commit_hash"),
            "mean_sharpe": cpcv_res["mean_sharpe"],
            "pbo": cpcv_res["pbo"],
            "dsr": dsr_res["dsr"],
            "merged_to_canary": merge_res["merged"],
            "merge_reason": merge_res.get("reason", "PROMOTED_TO_CANARY"),
        })

    return {
        "strategy_id": strategy_id,
        "autopsy_summary": autopsy["diagnoses"],
        "evaluated_branches": sprint_results,
    }
