# Source: MemLabs Lab 0 README
# Use Case: Brute-force XOR decoding of hex-encoded strings
# Prerequisites: Python 2 or 3 (with appropriate print syntax)

def xor_brute_force(hex_string):
    """Brute force XOR decode a hex string with all 255 possible keys."""
    import binascii
    a = binascii.unhexlify(hex_string)
    results = []
    for i in range(0, 255):
        b = ""
        for j in a:
            b = b + chr(j ^ i)
        results.append((i, b))
    return results

# Example usage from Lab 0:
hex_str = "335d366f5d6031767631707f"
decoded = xor_brute_force(hex_str)
for key, text in decoded:
    print(f"Key {key}: {text}")
# Flag found at key index 3: "1_4m_b3tt3r}"
